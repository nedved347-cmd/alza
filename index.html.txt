    <script>
        let counts = { returned: 0, canceled: 0, invalid: 0 };

        function changeCount(type, val) {
            counts[type] = Math.max(0, counts[type] + val);
            document.getElementById(type).innerText = counts[type];
        }

        function saveLog() {
            const isBoxFinished = document.getElementById('boxFinished').checked;
            const logEntry = {
                timestamp: new Date().toLocaleString('cs-CZ'),
                returned: counts.returned,
                canceled: counts.canceled,
                invalid: counts.invalid,
                finished: isBoxFinished
            };

            let logs = JSON.parse(localStorage.getItem('alza_logs') || '[]');
            logs.unshift(logEntry);
            localStorage.setItem('alza_logs', JSON.stringify(logs));

            // Reset
            counts = { returned: 0, canceled: 0, invalid: 0 };
            document.getElementById('returned').innerText = '0';
            document.getElementById('canceled').innerText = '0';
            document.getElementById('invalid').innerText = '0';
            document.getElementById('boxFinished').checked = false;

            renderLogs();
        }

        function renderLogs() {
            let logs = JSON.parse(localStorage.getItem('alza_logs') || '[]');
            const container = document.getElementById('logHistory');
            
            if (logs.length === 0) {
                container.innerHTML = 'Žádné záznamy.';
                return;
            }

            container.innerHTML = logs.map(l => `
                <div class="log-item">
                    <div class="log-date">${l.timestamp}</div>
                    <div>Vrácené: <b>${l.returned}</b> | Storno: <b>${l.canceled}</b> | Kódy: <b>${l.invalid}</b></div>
                    <div style="margin-top: 4px;">
                        Box: <span class="badge ${l.finished ? 'badge-success' : 'badge-danger'}">
                            ${l.finished ? 'HOTOVO' : 'NEDONOSENO'}
                        </span>
                    </div>
                </div>
            `).join('');
        }

        renderLogs();
    </script>
